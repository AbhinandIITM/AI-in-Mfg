\section{Introduction}

\begin{frame}{Problem: Dynamic Cutting Parameter Optimization}
    CNC face milling: rotating cutter removes material in multiple passes.
    \vspace{0.5em}

    \textbf{Parameters to optimize:}
    \begin{itemize}
        \item Spindle speed $n$ [RPM]
        \item Feed per revolution $f_r$ [mm/r]
        \item Axial depth of cut $a_p$ [mm]
    \end{itemize}

    \vspace{0.5em}
    \textbf{Four competing objectives:}
    \begin{itemize}
        \item \textbf{Productivity:} Maximize $\MRR = v_c \cdot a_p \cdot f_r$
        \item \textbf{Energy:} Minimize total electrical energy/part
        \item \textbf{Quality:} $\Ra \le 1.6\ \mu\text{m}$ (finished surface)
        \item \textbf{Tool Life:} $\VB \le 0.3\ \text{mm}$, avoid breakage
    \end{itemize}

    \vspace{0.5em}
    \alert{Traditional fixed parameters (handbooks, static NSGA-II) cannot adapt to tool wear, material variation, or geometry changes.}
\end{frame}

\begin{frame}{Proposed Approach}
    \footnotesize
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{1. Digital Twin (Physics)}
            \begin{itemize}
                \item Speed-dependent cutting energy $k_c(v_c) = k_{c0}(v_c/v_{\text{ref}})^m$
                \item Extended Taylor tool wear
                \item Empirical roughness regression
                \item Stochastic breakage
            \end{itemize}

            \textbf{2. DRL Agent (PPO)}
            \begin{itemize}
                \item One Gymnasium environment
                \item 11-D observation, context always present
                \item Energy-aware reward
                \item Conditioned policy ($w_2$ + material context)
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{3. Experimental Rigor}
            \begin{itemize}
                \item 100-scenario Monte Carlo + 95\% CIs
                \item Ablation: 7 weights $\times$ 3 seeds
                \item 5 classical baselines + receding-horizon
                \item Robustness: 3 regimes against the controller
                \item Regime map: the noise crossover against the controller
                \item Batch: one tool, ten parts
                \item Transfer experiment
                \item Full reproducibility
            \end{itemize}

            \textbf{4. Interactive Workstation}
            \begin{itemize}
                \item 2D stock depletion, micro-topography
                \item G-code generation, multi-controller comparison
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\section{Digital Twin Physics}

\begin{frame}{Key Physics Equations (MillingDT/config/physics.py)}
    \small
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \setlength{\jot}{2pt}
            \begin{align}
                v_c &= \frac{\pi D n}{1000} \\
                \MRR &= v_c \cdot a_p \cdot f_r \\
                t_{\text{cut}} &= \frac{L_{\text{cut}}}{n f_r} \\
                \Delta \VB &= C_{\text{wear}} v_c^{\alpha} f_r^{\beta} a_p^{\gamma} t_{\text{cut}} \\
                \Ra &= k_1 f_r^2 + k_2 \VB + \frac{k_3}{v_c} \\
                k_c(v_c) &= k_{c0} \left( \frac{v_c}{v_{\text{ref}}} \right)^m \\
                P_{\text{total}} &= P_{\text{idle}} + k_c(v_c) \cdot \MRR \\
                E_{\text{pass}} &= P_{\text{total}} \cdot t_{\text{cut}}
            \end{align}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{alertblock}{Key Insight: Interior Minimum}
            With $m=0.6 > 0$:
            \begin{itemize}
                \item Idle energy $\propto 1/(n f_r)$ $\downarrow$ with speed
                \item Cutting energy $\propto n^m$ $\uparrow$ with speed
                \item Part energy $E_{\text{part}}(v_c)$ has \textbf{interior minimum}
                \item Energy weight $w_2$ selects operating point
            \end{itemize}
            \end{alertblock}
            \vspace{0.2em}
            {\scriptsize
            Wear: $C_{\text{wear}} = 1.0\times10^{-2}$, anchored on a published tool life (about 9
            parts per tool): none of the three reference papers carries a Taylor wear model.}
        \end{column}
    \end{columns}
\end{frame}

\section{Environment Design}

\begin{frame}{One Gymnasium Environment: \texttt{MillingDTEnv-v1}}
    \footnotesize
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Two registrations}
            \begin{itemize}
                \item \texttt{env.ENV\_ID} = \texttt{MillingDTEnv-v1}, 200-step limit
                \item \texttt{env.BATCH\_ENV\_ID} = \texttt{MillingDTBatchEnv-v1}
                \item The energy weight and the material are \emph{inputs}, not separate tasks
            \end{itemize}

            \textbf{Observation (11-D, always the same shape)}
            \begin{itemize}
                \item 0--7: $\tanh$-squashed $\VB$, $\kappa_{\text{rem}}$, previous $n, f_r, a_p$,
                      $\MRR/\MRR_{\text{max}}$, $\tanh$-squashed $\Ra$, tool-broken flag
                \item 8--10: the context: $w_2$, $C_\text{wear}$ over $[0.6,1.4]$,
                      $k_1$ over $[0.8,1.2]$ of nominal
            \end{itemize}

            \textbf{Action:} $[-1,1]^3$ $\to$ clipped $\to$ machine envelope
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Reward per step}
            \setlength{\jot}{3pt}
            \begin{align}
                r = &\ 100\left(w_1 \frac{\MRR}{\MRR_\text{max}} - w_2 \frac{E_\text{pass}}{E_\text{ref}}\right)
                    + 20\,\Delta\!\left(\tfrac{\kappa_\text{removed}}{\kappa_\text{total}}\right) \\
                    &- 50\,\mathbb{1}[\VB > \VBSafe] - 100\,\mathbb{1}[\text{broken}] \\
                    &- 50\,\mathbb{1}[\text{finish} \wedge \Ra > \RaSafe] - 0.05
            \end{align}
            \vspace{0.2em}
            \begin{itemize}
                \item $+200$ when a part is finished in spec, plus a Pareto margin bonus of at most $+5$
                \item $w_1 = 1 - w_2$; safety limits $\VBSafe = 0.255$ mm and $\RaSafe = 1.36\ \mu$m
                \item breakage $p = 0.3 \cdot$ wear factor $\cdot$ (load factor)$^2$ above $\VB > 0.15$ mm
                \item \alert{the scale is a design decision}: the objective term is worth about 50 points per part, a quarter of a part
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Domain Randomization}
    \begin{itemize}
        \item \textbf{Workpiece:} $\kappa_{\text{total}} \in [2,15]$, $D \in [40,150]$, $L_{\text{cut}} \in [50,200]$
        \item \textbf{Material/Tool:} $C_\text{wear}$ at $[0.6,1.4]\times$ nominal, $k_1, k_2, k_3$ at $[0.8,1.2]\times$ nominal
        \item \textbf{Initial wear:} $V_{B,0} \sim \mathcal{U}[0, 0.8] V_{B,\text{max}}$
        \item \textbf{Energy weight:} drawn $\mathcal{U}[0,1]$ per episode under \texttt{--randomize-weight}
        \item \textbf{Shared tables:} same \texttt{MillingDT/config/scenarios.py} for training + Monte Carlo eval
        \item \smallskip
        \item \textbf{Measured:} the normalized physics features the policy sees on the training and the evaluation sets agree to 0.041 in the mean (std = 1), the same distribution
        \item \textbf{In the observation:} $C_\text{wear}$ and $k_1$ travel with the part, so the policy knows which material it is cutting
        \item \textbf{Belief and noise:} the context can carry a wrong material and the two measured features can carry noise, which is what the robustness experiment uses
    \end{itemize}
\end{frame}

\begin{frame}{Robustness: How to Read It}
    \scriptsize
    \begin{enumerate}
        \item \textbf{Inside a perfect digital twin the two methods are comparable, and the measured
        gap is partly an artifact of the reward.} The controller computes the optimum of the same
        objective at every pass with the exact model and the exact state, and it scores 248.9
        against the policy's 233.6. The gap is inside the noise: the standard error of the mean is
        about 8 for both arms, so a difference of 15 has a $t$ of about 1.4. And about nine of those
        fifteen points are the reward: the controller cuts the part in 5.85 passes against the
        policy's 4.00 and is paid for every pass.
        \item \textbf{The learned policy is the better throughput operator.} It removes \alert{22\% more
        material per minute} (1608 against 1318), finishes a part in \alert{a third less time} (11.4
        against 17.1 s) and spends 3.7\% less energy. It pays with twice the tool breakage (6\% against 3\%).
        \item \textbf{The advantage appears where a plant actually is.} With a material the controller
        has no model for, its reward collapses by three quarters, from 248.9 to 63.5, and the policy
        holds at 226.9. With a noisy wear measurement the controller falls to 181.3 and its spread
        doubles to 185.9 against the policy's 87.3, and the policy trained with the same noise keeps 213.6.
        The general policy scores 226.8 at that same 5\% noise.
        \item \textbf{The policy does not need the material to be told.} Told the truth it scores 215.1,
        against 226.9 for the policy that believes the nominal material.
    \end{enumerate}

    \vspace{0.1em}
    \begin{alertblock}{The claim}
    \scriptsize
    Inside a perfect digital twin the learned policy and the model-based controller are comparable:
    the policy is ahead on the physical metrics and behind on the breakage, and the gap the reward does
    show is partly the artifact of how the reward pays for a rate. The advantage of the learned policy
    appears where the controller's information advantage disappears, and the regime map on the next
    frames puts a number on it: the crossover is at \alert{about 2\% of the wear measurement's range}.
    The sensitivity of the comparison to the reward is published with the result.
    \end{alertblock}
\end{frame}

\begin{frame}{The Regime Map: How to Read It}
    \scriptsize
    \begin{enumerate}
        \item \textbf{The crossover is at about 2\% of the range of the wear measurement.} Below it the
        classical controller is the better operator, above it the learned policy is. That is the
        actionable form of the whole result.
        \item \textbf{The two degrade differently.} The controller falls from 248.9 to 171.5, a 31\% loss,
        because a noisy wear reading makes its plan violate the limits and break tools. The policy falls
        from 233.6 to 222.0, a 5\% loss, because a perturbed observation perturbs its action slightly.
        A model-based controller has a cliff, a learned policy has a slope.
        \item \textbf{Training for the noise does not pay.} The policy trained with the noise is below the
        general policy at every noise level, including the 5\% it was trained for, where it scores 213.6
        against 226.8. Its robustness comes from training across a distribution of scenarios.
    \end{enumerate}

    \vspace{0.2em}
    \begin{alertblock}{The fairness decisions}
    \scriptsize
    Both arms of the headline comparison are the general ones, neither is trained for the noise. The
    noise is tilted against the policy: the controller gets a noisy reading of the flank wear, the
    measurement it depends on, while the policy gets the noise on the flank wear \emph{and} on the
    remaining stock. The noise-trained policy is a third, clearly labelled arm and not the headline,
    because a controller can also be prepared for a noisy instrument by widening its safety margin, and
    that is a design choice rather than a training one. Of the two degraded regimes the noise test is
    the fairer one, because the noise is a property of the instrumentation and not of the method.
    \end{alertblock}
\end{frame}

\begin{frame}{Reward Sensitivity: How the Reward Pays for a Rate}
    \small
    The reward pays the removal rate of every pass, the standard way to express a rate in a reward
    that is paid per pass. The total credit therefore grows with the number of passes a plan takes,
    so a plan that cuts the same part in more, shallower passes is paid more for it.

    \vspace{0.1em}
    On the same 100 randomized scenarios at $w_2 = 0.4$, measured by
    \texttt{experiments/reward\_sensitivity.py}: the controller takes \alert{5.85 passes} against the
    policy's \alert{4.00} and collects \alert{27\% more objective credit} ($+0.413$ against $+0.326$)
    for the same material.

\begin{frame}{Reward Sensitivity: Three Checks and the Consequence}
    \footnotesize
    \begin{enumerate}
        \item \textbf{The objective scale is not the lever:} the controller leads at every scale from 5
        to 4000 and the gap only grows with it, 7.1 points at 5 and 352.0 at 4000, so the sign never flips.
        \item \textbf{The part level alternative was implemented and rejected:} it removes the pass count
        from the accounting, so the arms are much closer (5.0 points at the scale of 100, against 15.3)
        and the policy is ahead at the large scales (9.4 points at 1000, 57.4 at 4000). But the material a
        part gives up is fixed by its stock, so the throughput weight would be inert and the weight sweep
        would stop measuring what it claims to measure.
        \item \textbf{What a correct form would need:} a rate cannot be accumulated over a part without
        becoming a volume, so a reward has to either price the cycle time directly or compare the part's
        average rate to a reference at the end of the part. Both change the meaning of the weights, and the
        second makes the objective sparse and the credit assignment poor.
    \end{enumerate}

    \vspace{0.1em}
    \begin{alertblock}{The consequence}
    \footnotesize
    The ideal case reads as a \alert{tie, not a win}: the policy is ahead on the physical metrics (22\%
    more material removed, a third less cycle time, 3.7\% less energy) and behind on the tool breakage
    (6\% against 3\%), and the gap the reward does show is partly the reward's artifact. The reward as
    shipped carries it, documented rather than hidden.
    \end{alertblock}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
    \scriptsize
    \setlength{\itemsep}{0pt}
    \setlength{\topsep}{0pt}
    \setlength{\parsep}{0pt}
    \begin{itemize}
        \item \textbf{Digital Twin} with speed-dependent cutting energy, tool wear, roughness, breakage; the wear coefficient is anchored on a published tool life
        \item \textbf{Two Gymnasium Environments} (the part, 11-D, the batch, 13-D) and \textbf{One Production Policy} (1M steps): the energy weight is a runtime query
        \item \textbf{Rigorous Evaluation}: Monte Carlo (100 scenarios, 95\% CIs), ablation frontier, weight sweep against the 21 policies
        \item \textbf{Baselines}: five classical policies on the same 50 episodes; the receding-horizon controller is the strongest at $+253.5$
        \item \textbf{Robustness}: inside a perfect twin the two methods are comparable (248.9 against 233.6, the gap inside the noise and partly the reward's artifact); the controller collapses to 63.5 and 181.3, the policy holds at 226.9
        \item \textbf{Regime map}: the crossover is at \alert{about 2\% of the wear measurement's range}, the actionable form of the result
        \item \textbf{Reward sensitivity}: the credit grows with the number of passes (27\% more for the controller)
        \item \textbf{Batch}: 24\% fewer tool changes and 9.5\% less energy; 207.5 s against 113.1 s for the batch
        \item \textbf{Controller Initialization}: the result reversed, the initialization is now harmful (119.8 against 233.6)
        \item \textbf{Transfer Experiment}: the production policy needs no adaptation; the nominal-trained one does
        \item \textbf{Interactive Workstation} and \textbf{full reproducibility}: every result table from the committed artifacts by one script
    \end{itemize}

    \vspace{-1ex}
    \begin{alertblock}{Key Scientific Finding}
    \scriptsize
    The speed-dependent specific cutting energy creates a \textbf{genuine energy/throughput trade-off}, and DRL learns to navigate it pass-by-pass: across the frontier the removal rate falls \alert{41\%} while the energy of the part falls \alert{17\%}. Against a model-based controller the claim is a \alert{boundary}, not an advantage: inside a perfect digital twin the two are comparable, and the learned policy is the better operator once the wear measurement is noisier than \alert{about 2\% of its range}.
    \end{alertblock}
\end{frame}
